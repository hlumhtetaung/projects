1. Finding one time purchased orders

CREATE TABLE one_time_customers AS
SELECT *
FROM cleaned_retail_data
WHERE customer_id IN (
    SELECT customer_id
    FROM cleaned_retail_data
    GROUP BY customer_id
    HAVING COUNT(DISTINCT(order_id)) < 2
    );

one time orders: 28
total sales from one time orders: 5181.76

2. Finding standard (2-10 times purchased) orders

CREATE TABLE standard_customers AS
SELECT *
FROM cleaned_retail_data
WHERE customer_id IN (
    SELECT customer_id
    FROM cleaned_retail_data
    GROUP BY customer_id
    HAVING COUNT(DISTINCT(order_id)) >= 2 AND COUNT(DISTINCT(order_id)) <= 10
    );

standard orders: 8707
total sales from standard orders: 2020562.20
